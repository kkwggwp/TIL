> server.js

```javascript
router.post('/hobby', (req, res)=> {
    const { hobby } = req.body;
    console.log(hobby);
    // hobby가 존재하는지 확인하고, 항상 배열 형태가 되도록 처리합니다.
    // 1. hobby가 undefined이면 빈 배열 []
    // 2. hobby가 배열이 아니면(문자열이면) [hobby]
    // 3. hobby가 배열이면 그대로 사용
    const hobbies = hobby ? (Array.isArray(hobby) ? hobby : [hobby]) : [];
    
    res.render('hobby', { hobby: hobbies });
});
```
