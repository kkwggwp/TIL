## 1. 콜백 방식 (mysql2 모듈)
```javascript
// DB와 연결 할 수 있는 모듈
const mysql = require('mysql2')

// DB 연결 정보 입력
let conn = mysql.createConnection({
    host: '127.0.0.1',
    user: 'testuser',
    password: '12345',
    port: 3306,
    database: 'mynode'
});

//createConnection을 사용시 conn.end() 처럼 연결을 닫아줘야함
// DB 연결 통로 생성
conn.connect();

router.post('/login' (req, res) => {
    console.log("/login 접근 확인");
    console.log(req.body);
    let id = req.body.id;
    let pw = req.body.pw;
    sql = 'SELECT * FROM member WHERE id = ?';
    params = [id];

    conn.query(sql, params, (err, rows) => {
        if(!err){
            if(rows.length > 0){
                console.log('조회 성공');
                let dbId = rows[0].id;
                let dbPw = rows[0].pw;
                if (id === dbId && pw === dbPw){
                    console.log('login succesful');
                    res.redirect('http://localhost:5500/express02/public/success.html');
                }
                else{
                    console.log('login failed');
                    res.redirect('http://localhost:5500/express02/public/fail.html');
                }
            }
            else{
                console.log('조회 실패');
                res.status(404).send(`<h1>user not found</h1>`);
            }
        }
        else{
            console.log('db 연결 실패');
        }
    });    
});
```
1. conn.query() 를 사용한 콜백 방식은 코드가 중첩되어 흐름을 파악하기 어려움
2. 비동기 작업의 결과를 콜백 함수 내에서만 다룰 수 있음
3. 전통적이지만 복잡한 로직에서는 잘 사용되지 않음
4. try catch는 동기적이므로 이러한 비동기적 로직에서는 에러를 잡을 수 없음
5. mysql2 모듈과 createConnection() 방식은 느리고 비효율적이라 잘 사용되지 않음
6. ㄴ db 에러도 출력이 안됨

## 2. try catch 사용한 async/await 방식 (mysql2/promise)
```javascript
const mysql = require('mysql2/promise');

// Connection Pool 생성
const pool = mysql.createPool({
    host: '127.0.0.1',
    user: 'testuser',
    password: '12345',
    port: 3306,
    database: 'mynode',
    waitForConnections: true,
    connectionLimit: 10,
    queueLimit: 0
});

module.exports = pool;
```
> db.js
```javascript
const pool = require('../config/db'); // DB 커넥션 풀 가져오기
const path = require('path');

router.post('/login', async (req, res) => {
    console.log("/login 접근 확인");
    console.log(req.body);

    const { id, pw } = req.body;
    const sql = 'SELECT * FROM member WHERE id = ?';

    try {
        const [rows] = await pool.query(sql, [id]);

        if (rows.length > 0) {
            console.log('id 조회 성공');
            const dbPw = rows[0].pw;

            if (pw === dbPw) {
                console.log('login successful');
                res.send(`<h1>login successful</h1>`);
            } else {
                console.log('login failed: incorrect password');
                res.send(`<h1>incorrect password</h1>`);
            }
        } else {
            console.log('login failed: user not found');
            res.status(404).send(`<h1>user not found</h1>`);
        }
    } catch (error) {
        console.error('DB 쿼리 실패:', error);
        res.status(500).send(`<h1>Internal Server Error</h1>`);
    }
});
```
> login.js

1. 동기 코드처럼 위에서 아래로 읽혀 가독성이 좋다
2. try catch를 구문을 사용해 일관성 있게 처리 가능
3. await를 통해 비동기 작업의 결과를 변수에 바로 할당받아 사용 가능
4. 현대 자바스크립트에서 비동기 처리를 위한 표준이자 가장 권장하는 방식
5. createPool은 미리 정해진 갯수의 커넥션을 생성해두고 필요할때마다 빌려쓰고 반납함
6. ㄴ 빠르고 효율적이라 웹 서버, API 서버등 웬만한곳에서 다 쓰임 db 에러도 출력됨

# 결론 : await/async를 mysql2/promise랑 잘 버무려서 써라
