## 1. session 추가
```javascript
const session = require('express-session');
// 세션 생성
app.use(session({
    // secret : 보안 키 설정
    secret : 'my-secret',
    // resave : 세션에 따른 요청마다 다시 저장할건지
    // false : 세션값이 바뀔때만 저장하겠다
    resave : false,
    // saveUnitinitialized : 세션에 아무 데이터가 없더라도 사용여부 선택하는 속성
    saveUninitialized : false
}));

// 세션 사용
app.use((req, res, next) => {
    res.locals.session = req.session;
    // console.log(req.session);
    // 다음 코드 실행
    next();
});
```

## 2. 세션 활용한 메시지 기능
```javascript
// 메시지 전송
router.post('/sendMessage', async (req, res) => {
    const {name, email, message} = req.body;

    const sql = `INSERT INTO message (sendName, sendEmail, sendMessage, sendTime) VALUES (?, ?, ?, ?)`
    const userEmail = req.session.user.email;

    if (!req.session.user) {
        console.log('메시지 전송 실패 : 로그인 필요');
        return res.send('<script>alert("로그인이 필요합니다."); history.back();</script>');
    }

    if (email === userEmail) {
        console.log(`${userEmail} 메시지 전송 실패 : 동일한 유저`);
        return res.send('<script>alert("자신에게 메시지를 보낼 수 없습니다."); history.back();</script>');
    }

    try {
        const [result] = await pool.query(sql, [name, email, message, new Date()]);
        if (result.affectedRows > 0) {
            console.log(`${userEmail}메시지 전송 성공`);
            // res.redirect('/');
            res.send('<script>alert("메시지 전송 성공"); location.href="/";</script>')
        }        
    } catch (error) {
        console.error('DB 쿼리 실패', error);
        res.status(500).send('Internal Server Error');
    }
});

// 메시지 확인
router.get('/board', async (req, res) => {
    if (!req.session.user) {
        return res.send('<script>alert("로그인이 필요합니다."); location.href="/";</script>');
    };

    const userEmail = req.session.user.email;
    const sql = 'SELECT * FROM message WHERE sendEmail = ? ORDER BY sendTime DESC';

    try {
        const [messages] = await pool.query(sql, [userEmail]);
        console.log(`${userEmail} 메시지 확인 성공`);
        res.render('board', { messages: messages });
    } catch (error) {
        console.error('DB 쿼리 실패', error);
        res.status(500).send('Internal Server Error');
    };
});
```
