```python
# ==============================================================================
# 1. 라이브러리 임포트
# ==============================================================================
import asyncio
import os
import uuid
import aiofiles
import json
from fastapi import FastAPI, UploadFile, WebSocket, WebSocketDisconnect, Form, Body
from fastapi.responses import HTMLResponse, JSONResponse
from fastapi.staticfiles import StaticFiles
from . import processing
from typing import List, Dict

# ==============================================================================
# 2. FastAPI 앱 초기화 및 파일 설정
# ==============================================================================
app = FastAPI()

app.mount("/static", StaticFiles(directory="templates"), name="static")

TEMP_VIDEO_DIR = "temp_videos"
HISTORY_FILE = "history.json"
REPORTS_FILE = "reports.json"

os.makedirs(TEMP_VIDEO_DIR, exist_ok=True)
if not os.path.exists(HISTORY_FILE):
    with open(HISTORY_FILE, "w", encoding='utf-8') as f:
        json.dump([], f)
if not os.path.exists(REPORTS_FILE):
    with open(REPORTS_FILE, "w", encoding='utf-8') as f:
        json.dump([], f)

# ==============================================================================
# 3. API 엔드포인트
# ==============================================================================
@app.get("/", response_class=HTMLResponse)
async def read_root():
    """메인 HTML 페이지를 서빙합니다."""
    with open("templates/index.html") as f:
        return HTMLResponse(content=f.read(), status_code=200)

@app.post("/upload")
async def upload_video(cam_id: int = Form(...), file: UploadFile = UploadFile(...)):
    """비디오 파일을 임시 폴더에 업로드합니다."""
    file_path = os.path.join(TEMP_VIDEO_DIR, f"{uuid.uuid4()}_{file.filename}")
    async with aiofiles.open(file_path, "wb") as buffer:
        content = await file.read()
        await buffer.write(content)
    return {"message": "업로드 성공", "path": file_path, "cam_id": cam_id}

@app.get("/history", response_class=JSONResponse)
async def get_history():
    """탐지 기록(history.json)을 반환합니다."""
    try:
        async with aiofiles.open(HISTORY_FILE, "r", encoding='utf-8') as f:
            content = await f.read()
            return JSONResponse(content=json.loads(content))
    except (FileNotFoundError, json.JSONDecodeError):
        return JSONResponse(content=[])

@app.post("/generate_report")
async def generate_report(logs: List[Dict] = Body(...)):
    """클라이언트로부터 받은 탐지 기록을 기반으로 AI 보고서를 생성합니다."""
    try:
        report_text = await processing.generate_ai_report(logs)
        return JSONResponse(content={"report": report_text})
    except Exception as e:
        return JSONResponse(status_code=500, content={"detail": f"보고서 생성 중 서버 오류 발생: {e}"})

@app.post("/save_report")
async def save_report(report_data: dict = Body(...)):
    """클라이언트로부터 받은 보고서 데이터를 reports.json에 저장합니다."""
    try:
        with open(REPORTS_FILE, "r+", encoding='utf-8') as f:
            reports = json.load(f)
            reports.insert(0, report_data)
            f.seek(0)
            f.truncate()
            json.dump(reports, f, indent=4, ensure_ascii=False)
        return JSONResponse(content={"message": "보고서가 성공적으로 저장되었습니다."})
    except Exception as e:
        return JSONResponse(status_code=500, content={"detail": f"보고서 저장 중 오류 발생: {e}"})

@app.get("/reports")
async def get_reports():
    """저장된 보고서(reports.json) 목록을 반환합니다."""
    try:
        async with aiofiles.open(REPORTS_FILE, "r", encoding='utf-8') as f:
            content = await f.read()
            return JSONResponse(content=json.loads(content))
    except (FileNotFoundError, json.JSONDecodeError):
        return JSONResponse(content=[])

@app.websocket("/ws")
async def websocket_endpoint(websocket: WebSocket):
    """WebSocket 연결을 처리하고 영상 분석을 시작합니다."""
    await websocket.accept()
    video_processor = processing.VideoProcessor(websocket)
    try:
        while True:
            data = await websocket.receive_json()
            await video_processor.handle_client_message(data)
    except WebSocketDisconnect:
        print("클라이언트 연결이 끊어졌습니다.")
    except Exception as e:
        print(f"WebSocket 오류 발생: {e}")
    finally:
        video_processor.cleanup()
        print("WebSocket 연결이 종료되고 리소스가 정리되었습니다.")
```
