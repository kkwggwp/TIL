class VideoProcessor:
    def __init__(self, websocket):
        self.websocket = websocket
        self.video_paths = {}
        self.caps = {}
        self.analysis_data = {}
        self.settings = {'confidence': 0.35, 'iou': 0.45, 'frame_skip_rate': 5, 'detection_confirm_count': 3}
        self.processing_tasks = {}
        
        try:
            self.llm = ChatOpenAI(temperature=0.7, model_name="gpt-4o-mini")
        except Exception as e:
            print(f"ChatOpenAI 모델 초기화 실패: {e}")
            self.llm = None

    async def process_single_video(self, cam_id):
        cap = self.caps.get(cam_id)
        if not cap: return
        frame_counter = 0
        while cap.isOpened():
            ret, frame = cap.read()
            if not ret:
                cap.set(cv2.CAP_PROP_POS_FRAMES, 0)
                continue
            
            original_frame = frame.copy() # [신규] LLM 전송을 위해 원본 프레임 복사
            frame = cv2.resize(frame, TARGET_RESOLUTION)
            display_frame = frame
            data = self.analysis_data.get(cam_id)
            is_cooldown = data and time.time() < data.get('cooldown_until', 0)

            if not is_cooldown:
                if frame_counter % self.settings['frame_skip_rate'] == 0:
                    annotated_frame, labels, boxes_abs = self.analyze_frame(frame)
                    detection_result = self.update_detection_state(cam_id, labels, boxes_abs)
                    
                    if detection_result:
                        # [수정] 1차 경보 전송 후, 즉시 LLM 2차 판단 시작
                        await self.send_update(detection_result)
                        await self.send_update({'type': 'llm_analysis_started', 'cam_id': cam_id})
                        asyncio.create_task(self._get_llm_judgment(cam_id, original_frame))
                        
                    display_frame = annotated_frame
            else:
                if data['alert_level'] != 'normal':
                    data['alert_level'] = 'normal'
                    await self.send_update({'type': 'status_update', 'cam_id': cam_id, 'alert_level': 'normal'})

            _, buffer = cv2.imencode('.jpg', display_frame)
            frame_b64 = base64.b64encode(buffer).decode('utf-8')
            await self.send_update({'type': 'video_frame', 'cam_id': cam_id, 'frame': frame_b64})
            frame_counter += 1
            await asyncio.sleep(1/30)

    def analyze_frame(self, frame):
        # ... (이하 코드 변경 없음)
        results = model.predict(frame, conf=self.settings['confidence'], iou=self.settings['iou'], verbose=False, device=device)
        annotated_frame = results[0].plot()
        labels = [model.names[int(cls)] for cls in results[0].boxes.cls]
        valid_indices = [i for i, label in enumerate(labels) if label in ["Fire", "Smoke"]]

        filtered_boxes_absolute = results[0].boxes.xyxy[valid_indices].cpu().numpy()

    def update_detection_state(self, cam_id, labels, boxes_abs):
        # ... (이하 코드 변경 없음)
        data = self.analysis_data.get(cam_id)
        if not data: return None
        if not labels:
            is_pending = data['alert_level'] == 'pending'
            time_since_last_detection = time.time() - data.get('last_detected_time', 0)
            if is_pending and time_since_last_detection < FLICKER_THRESHOLD_SECONDS:
                return None
            if data['alert_level'] != 'confirmed' and data['alert_level'] != 'normal':
                data.update({'detection_counter': 0, 'alert_level': 'normal'})
                return {'type': 'status_update', 'cam_id': cam_id, 'alert_level': 'normal'}
            data['detection_counter'] = 0
            return None
        data['detection_counter'] += 1
        data['last_detected_time'] = time.time()
        data['bbox_history'].append(boxes_abs)
        data['labels'] = labels
        current_fire_level = self.determine_fire_level(labels, data['bbox_history'])
        if data['alert_level'] == 'normal':
            if data['detection_counter'] >= self.settings['detection_confirm_count']:
                data['alert_level'] = 'pending'
                data['fire_level'] = current_fire_level
                return {'type': 'detection_alert', 'cam_id': cam_id, 'location': CAM_LOCATIONS[cam_id], 'alert_level': 'pending', 'fire_level': current_fire_level}
        elif data['alert_level'] == 'pending':
            if current_fire_level == '2단계' and data.get('fire_level') == '1단계':
                data['fire_level'] = '2단계'
                return {'type': 'detection_alert', 'cam_id': cam_id, 'location': CAM_LOCATIONS[cam_id], 'alert_level': 'pending', 'fire_level': '2단계'}
        return None

    async def handle_client_message(self, data):
        action = data.get("action")
        cam_id = data.get("cam_id")

        if action == "start_analysis":
            path = data.get("path")
            if cam_id not in self.caps:
                self.caps[cam_id] = cv2.VideoCapture(path)
                self.video_paths[cam_id] = path
                self.analysis_data[cam_id] = {
                    'bbox_history': deque(maxlen=10), 'alert_level': 'normal', 
                    'detection_counter': 0, 'cooldown_until': 0, 'last_detected_time': 0,
                    'human_intervened': False  # [신규] 관리자 개입 플래그 초기화
                }
                task = asyncio.create_task(self.process_single_video(cam_id))
                self.processing_tasks[cam_id] = task
                print(f"[{cam_id}번 채널] 분석 시작: {path}")

        elif action == "confirm_alert":
            if self.analysis_data.get(cam_id):
                self.analysis_data[cam_id]['human_intervened'] = True # [신규] 관리자 개입 플래그 설정
                self.analysis_data[cam_id]['alert_level'] = 'confirmed'
                response_data = await self.generate_langchain_messages(CAM_LOCATIONS[cam_id], self.analysis_data[cam_id]['fire_level'], self.analysis_data[cam_id]['labels'])
                await self.save_alert_to_history(cam_id, self.analysis_data[cam_id], response_data)
                await self.send_update({'type': 'confirmed_alert_details', 'cam_id': cam_id, 'details': response_data})

        elif action in ["false_alarm", "resolve_alert"]:
            if self.analysis_data.get(cam_id):
                self.analysis_data[cam_id]['human_intervened'] = True # [신규] 관리자 개입 플래그 설정
                self.analysis_data[cam_id].update({'alert_level': 'normal', 'detection_counter': 0, 'cooldown_until': time.time() + COOLDOWN_SECONDS})
                await self.send_update({'type': 'status_update', 'cam_id': cam_id, 'alert_level': 'normal'})

        elif action == "update_settings":
            self.settings.update(data.get("payload", {}))

    def determine_fire_level(self, labels, bbox_history):
        """[최종 개선] 2단계 화재 판단 기준을 더욱 엄격하고 안정적으로 조정합니다."""
        def get_area(boxes):
            if boxes is None or len(boxes) == 0: return 0
            return sum((box[2] - box[0]) * (box[3] - box[1]) for box in boxes)

        if not bbox_history: return "1단계"
        
        current_boxes = bbox_history[-1]
        current_area = get_area(current_boxes)
        screen_area = TARGET_RESOLUTION[0] * TARGET_RESOLUTION[1]

        # --- 2단계 판단 로직 (우선순위 순) ---

        # 1순위: 절대 크기 기준 (매우 큰 화재)
        # 개별 불꽃의 크기가 화면의 8%를 초과하는가?
        if current_boxes is not None:
            for box in current_boxes:
                box_area = (box[2] - box[0]) * (box[3] - box[1])
                if box_area / screen_area > 0.08: # 기준을 8%로 상향
                    return '2단계'

        # 2순위: 복합 위험 기준 (연기를 동반한 화재)
        # '불꽃'과 '연기'가 동시에 탐지되었는가?
        if 'Fire' in labels and 'Smoke' in labels:
            return '2단계'

        # 3순위: 의미 있는 확산 기준 (성장하는 화재)
        if len(bbox_history) > 1 and bbox_history[-2] is not None:
            prev_area = get_area(bbox_history[-2])
            
            # 불이 이미 화면의 2% 이상을 차지하고 있으면서,
            # 그 크기가 이전보다 1.5배(50%) 이상 커졌는가?
            if prev_area > (screen_area * 0.02) and (current_area / prev_area) > 1.5:
                return '2단계'
            
        # 위 모든 조건에 해당하지 않으면 1단계
        return '1단계'

    async def save_alert_to_history(self, cam_id, analysis_data, response_data):
        history_entry = {"timestamp": datetime.now().strftime("%Y-%m-%d %H:%M:%S"), "location": CAM_LOCATIONS[cam_id], "fire_level": analysis_data.get('fire_level'), "detected_objects": list(set(analysis_data.get('labels', []))), "messages": response_data}
        try:
            with open(HISTORY_FILE, "r+", encoding='utf-8') as f:
                history = json.load(f)
                history.insert(0, history_entry)
                f.seek(0)
                f.truncate()
                json.dump(history, f, indent=4, ensure_ascii=False)
        except (FileNotFoundError, json.JSONDecodeError):
            with open(HISTORY_FILE, "w", encoding='utf-8') as f:
                json.dump([history_entry], f, indent=4, ensure_ascii=False)
        except Exception as e:
            print(f"히스토리 저장 오류: {e}")

    async def send_update(self, data):
        await self.websocket.send_json(data)

    def cleanup(self):
        for task in self.processing_tasks.values():
            if not task.done():
                task.cancel()
        for cap in self.caps.values():
            cap.release()
        for path in self.video_paths.values():
            if os.path.exists(path):
                try:
                    os.remove(path)
                except OSError as e:
                    print(f"임시 파일 삭제 오류: {e}")
        print("모든 채널의 처리 중단 및 임시 파일 정리 완료.")
