```javascript
// 회원가입
router.post('/signUp', async (req, res) => {
    console.log(req.body);
    const { id, pw, email, phone, address } = req.body;
    const sql = 'INSERT INTO member2 (id, pw, email, phone, address) VALUES (?, ?, ?, ?, ?)';
    
    try {
        const [result] = await pool.query(sql, [id, pw, email, phone, address]);
        if (result.affectedRows > 0) {
            console.log('회원가입 성공');
            res.render('index', { signUp : 'success' });
        } else {
            console.log('회원가입 실패');
            res.render('index', { signUp : 'fail' });
        }
    } catch (error) {
        console.error("DB 쿼리 실패",error);
        res.status(500).send('Internal Server Error');
    }
});

// 로그인
router.post('/login', async (req, res) => {
    const {id, pw} = req.body;
    const sql = 'SELECT * FROM member2 WHERE id = ?';

    try {
        const [rows] = await pool.query(sql, [id]);

        // 1. ID가 존재하지 않거나, 2. 비밀번호가 일치하지 않는 경우
        if (rows.length === 0 || pw !== rows[0].pw) {
            console.log('로그인 실패: 아이디 또는 비밀번호 불일치');
            // 실패 원인을 구분하지 않고 항상 동일한 응답을 보냅니다.
            return res.render('index', { loginStatus: 'fail' });
        }

        // 위 조건을 모두 통과한 경우에만 성공으로 처리합니다.
        console.log('로그인 성공');
        res.render('index', { loginStatus: 'success' });
    } catch (error) {
        console.error("DB 쿼리 실패", error);
        res.status(500).send('Internal Server Error');
    }
});
```

> index.ejs

1. `signUp`과 `loginStatus`는 따로 선언하지 않았기 때문에 기본적으로 `undefined` 상태라 그걸 이용하면 따로 선언하지 않아도 됨
2. ㄴ 백엔드에서 프론트 코드를 삽입하는것보다 위와 같은 경우가 백엔드와 프론트가 잘 나누어져서 효율적인 구조
3. 로그인 오류 메시지는 `enumeration` 공격으로 `id`를 알아 낼 수 있으므로 사용자가 없음 같은 메시지는 없는게 좋음
4. ㄴ 같은 맥락으로 아이디 또는 비밀번호가 올바르지 않는다는 메시지만 출력하는게 좋음
