## 1. /update
```javascript
// 회원정보 수정
router.post('/update', async (req, res) => {
    const { email, id, pw, phone, address } = req.body;

    const updates = {};
    if (id) updates.id = id;
    if (pw) updates.pw = pw;
    if (phone) updates.phone = phone;
    if (address) updates.address = address;

    if (Object.keys(updates).length === 0) {
        return res.send('<script>alert("수정할 내용이 없습니다."); history.back();</script>');
    }

    const sql = 'UPDATE member2 SET ? WHERE email = ?';

    try {
        await pool.query(sql, [updates, email]);
        console.log(`${email} 회원정보 수정 성공`);
        res.redirect('/');
    } catch (error) {
        console.error("DB 쿼리 실패", error);
        res.status(500).send('Internal Server Error');
    }
});
```
- if문으로 필요한것만 실행, Object.keys로 객체의 모든 key를 배열로 반환
