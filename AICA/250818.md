## 2. 배열 및 배열 함수
```

// 배열 생성
let empty5 = new Array(5); // 권장하지 않음
let arr = []; // 배열 리터럴을 사용하는게 훨씬 명확함

// 인덱스를 지정해 값 추가
fruits[3] = "딸기" // 존재하지 않는 인덱스에 직접 값을 할당하는건 중간에 비어있는 공간을 만들 수 있음
fruits.push("딸기") // 의도를 명확히 하는게 안전함

// 원본 배열을 직접 변경하는 메서드
splice() // 데이터의 불변성을 중요하게 여기니 가급적 사용 X slice()나 filter() 사용 권장
reverse() // 사용하고 싶으면 원본 배열을 복사 후 실행하라

// 요소의 존재 여부 확인
fruits.indexOf("딸기") // -1이 아니면 존재한다고 판단
fruits.includes("딸기") // true or false 반환해 가독성이 훨씬 좋음

// for 반복문 대안
for (let fruit of fruits) // 인덱스 없이 배열의 값만 필요할때 가장 깔끔
fruits.forEach(fruit => {}) // 각 요소를 순회하면 특정 작업을 할때 사용
