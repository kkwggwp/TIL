# nodeJS 모듈화
## server.js
```javascript
const express = require('express');
// app : 서버 정보를 모두 가지고 있는 변수
const app = express();

// post 방식으로 데이터 통신시 body영역을 사용할 수 있게 허용
app.use(express.urlencoded({extended: true}));
app.use(express.json())

// GET으로 요청 들어오면 콜백 함수로 응답
app.get('/', (req, res) => {
    console.log("접근 확인");

});

const dan = require('./router/dan');
app.use(dan);

const plus = require('./router/plus');
app.use(plus);

const user = require('./router/user');
app.use(user);

app.listen(3000);
```
## user.js
```javascript
const express = require('express');
const router = express.Router();

// router : 서버에 적용 되는 기능의 모듈화 변수

router.post('/login', (req, res) => {
    console.log("login 접근 확인");
    console.log(req.body);

    let id = req.body.id;
    let pw = req.body.pw;

    if(id == '1' && pw == '123'){
        res.write('success');
    } else {
        res.write('fail');
    };

    res.end();
});

module.exports = router;
```
