## 1. 허프 선 변환
### 1.1 허프 선 변환
`lines = cv2.HoughLines(img, rho, theta, threshold, lines, srn=0, stn=0 min_theta, max_theta)`
```python
lines = cv2.HoughLines(edges, 1, np.pi / 180, 130)
for line in lines:
    r, theta = line[0] # 거리와 각도 w, h
    tx, ty = np.cos(theta), np.sin(theta) # x, y축에 대한 삼각비 활용
    x0, y0 = tx * r, ty * r # x, y 평면에서의 절편
    # 기준 좌표에 빨간원
    cv2.circle(img2, (int(abs(x0)), int(abs(y0))), 3, (0, 0, 225), -1)
    # 직선의 방정식으로 그리기 위한 두개의 점(시작점과 끝점)
    x1, y1 = int(x0 + w * (-ty)), int(y0 + h*tx)
    x2, y2 = int(x0 - w * (-ty)), int(y0 - h*tx)
    # 선 그리기
    cv2.line(img2, (x1, y1), (x2, y2), (0, 255, 0), 1)
```
